POST    /api/blood-requests	  			                Crear solicitud
GET	    /api/blood-requests	  			                Listar con filtros
GET	    /api/blood-requests/:id			                Obtener por id
PATCH	  /api/blood-requests/:id			                Actualizar
DELETE	/api/blood-requests/:id			                Eliminar
GET	    /api/blood-requests/:id/compatible-donors	  Donantes compatibles
POST	  /api/donors					                        Crear donante
GET	    /api/donors					                        Listar con filtros
GET	    /api/donors/compatible?recipientType=O-		  Buscar compatibles
GET	    /api/donors/:id					                    Obtener por id
PATCH	  /api/donors/:id					                    Actualizar
DELETE	/api/donors/:id					                    Eliminar
